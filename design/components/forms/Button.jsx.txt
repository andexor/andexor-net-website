import React from 'react';

/**
 * Andexor primary action control. Renders a <button> (or any element via `as`).
 */
export function Button({
  variant = 'primary',
  size = 'md',
  block = false,
  onInk = false,
  leftIcon,
  rightIcon,
  as: Tag = 'button',
  className = '',
  children,
  ...rest
}) {
  const cls = [
    'an-btn',
    `an-btn--${variant}`,
    size === 'sm' && 'an-btn--sm',
    size === 'lg' && 'an-btn--lg',
    block && 'an-btn--block',
    onInk && 'an-btn--on-ink',
    className,
  ].filter(Boolean).join(' ');

  return (
    <Tag className={cls} {...rest}>
      {leftIcon}
      {children}
      {rightIcon}
    </Tag>
  );
}
