import React from 'react';

/** Avatar with image or initials fallback. */
export function Avatar({ src, name = '', size = 'md', className = '', ...rest }) {
  const initials = name
    .split(' ')
    .filter(Boolean)
    .slice(0, 2)
    .map((w) => w[0])
    .join('')
    .toUpperCase();
  return (
    <span className={['an-avatar', `an-avatar--${size}`, className].filter(Boolean).join(' ')} {...rest}>
      {src ? <img src={src} alt={name} /> : initials}
    </span>
  );
}
