import React from 'react';

/** Compact status / category label. */
export function Badge({ tone = 'neutral', dot = false, className = '', children, ...rest }) {
  const cls = [
    'an-badge',
    `an-badge--${tone}`,
    dot && 'an-badge--dot',
    className,
  ].filter(Boolean).join(' ');
  return <span className={cls} {...rest}>{children}</span>;
}
